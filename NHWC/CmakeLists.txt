
cmake_minimum_required(VERSION 3.16)

project(NHWC LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(NHWC
    main.cpp

    Patterns.hpp
    CacheTests.hpp
    CsvWriter.hpp

    strategies/Cache.hpp
    strategies/lfu_hands.hpp
    strategies/arc_hands.hpp
    strategies/LIRC_hands.hpp
    strategies/2q_hands.hpp
    strategies/BeladyCache.hpp
    strategies/TwoLevelCache.hpp
)

target_include_directories(NHWC PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/strategies
)
